cmake_minimum_required(VERSION 3.24)
project(ETLForge VERSION 42.0.0.0 LANGUAGES CXX)
get_filename_component(ETLFORGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
list(PREPEND CMAKE_PREFIX_PATH
  "${ETLFORGE_ROOT}/Tools/vcpkg/installed/x64-windows"
  "${ETLFORGE_ROOT}/Dependencies/windows-x64")
if(DEFINED ENV{ETLFORGE_PYTHON} AND NOT "$ENV{ETLFORGE_PYTHON}" STREQUAL "")
  set(Python3_EXECUTABLE "$ENV{ETLFORGE_PYTHON}" CACHE FILEPATH "Python interpreter" FORCE)
endif()
if(NOT Python3_EXECUTABLE)
  find_program(Python3_EXECUTABLE NAMES python.exe python3.12.exe python3 python
    HINTS "C:/Tools/aider-env/Scripts" "$ENV{USERPROFILE}/.local/bin")
endif()
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
if(NOT WIN32)
  message(FATAL_ERROR "ETLForge currently targets Windows only")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ETLForge supports 64-bit builds only")
endif()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(ETLFORGE_COMMON "${CMAKE_CURRENT_SOURCE_DIR}/src/native/common")
set(ETLFORGE_INTEGRATION "${CMAKE_CURRENT_SOURCE_DIR}/src/native/integration")
add_library(etlforge_tokenizer STATIC
  src/native/common/konga/Utils/StringTokenizer.cpp)
target_include_directories(etlforge_tokenizer PUBLIC "${ETLFORGE_COMMON}")
target_compile_definitions(etlforge_tokenizer PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(etlforge_tokenizer PRIVATE /W4 /EHsc)
endif()
add_library(etlforge_vectorutils STATIC src/native/common/konga/Utils/VectorUtils.cpp)
target_include_directories(etlforge_vectorutils PUBLIC "${ETLFORGE_COMMON}")
add_library(etlforge_memutils STATIC src/native/common/konga/Utils/MemUtils.cpp)
target_include_directories(etlforge_memutils PUBLIC "${ETLFORGE_COMMON}")
target_compile_definitions(etlforge_memutils PUBLIC _WIN32_WINNT=0x0602 NOMINMAX)
if(BUILD_TESTING)
  add_executable(etlforge_memutils_tests tests/native/memutils_tests.cpp)
  target_link_libraries(etlforge_memutils_tests PRIVATE etlforge_memutils)
  add_test(NAME native.memutils COMMAND etlforge_memutils_tests)
  add_executable(etlforge_vectorutils_tests tests/native/vectorutils_tests.cpp)
  target_link_libraries(etlforge_vectorutils_tests PRIVATE etlforge_vectorutils)
  add_test(NAME native.vectorutils COMMAND etlforge_vectorutils_tests)
  add_executable(etlforge_tokenizer_tests tests/native/tokenizer_tests.cpp)
  target_link_libraries(etlforge_tokenizer_tests PRIVATE etlforge_tokenizer)
  add_test(NAME native.tokenizer COMMAND etlforge_tokenizer_tests)
endif()
# This opt-in target probes the complete legacy utility project. It is not a release target.
include(cmake/LegacyUtils.cmake)



# Validate the production XML converter against the rebuilt x64 Xerces runtime.
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../Dependencies/windows-x64")
find_package(XercesC 3.2 QUIET)
if(BUILD_TESTING AND XercesC_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../Dependencies/boost_1_86_0/boost/version.hpp")
  add_executable(etlforge_xmlstrings_tests tests/native/xmlstrings_tests.cpp)
  target_include_directories(etlforge_xmlstrings_tests PRIVATE
    "${ETLFORGE_COMMON}" "${CMAKE_CURRENT_SOURCE_DIR}/../Dependencies/boost_1_86_0")
  target_link_libraries(etlforge_xmlstrings_tests PRIVATE XercesC::XercesC)
  add_test(NAME native.xmlstrings COMMAND etlforge_xmlstrings_tests)
  set_tests_properties(native.xmlstrings PROPERTIES ENVIRONMENT_MODIFICATION
    "PATH=path_list_prepend:${CMAKE_CURRENT_SOURCE_DIR}/../Dependencies/windows-x64/bin")
endif()
