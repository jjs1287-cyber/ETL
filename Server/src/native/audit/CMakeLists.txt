cmake_minimum_required(VERSION 3.27)
project(ETLForgeAudit LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "ETLForge audit requires Windows x64")
endif()
set(ETLFORGE_BOOST_ROOT "C:/ETL/Dependencies/boost_1_86_0" CACHE PATH "Boost headers")
add_library(etlforge_audit STATIC AuditComparison.cpp AuditDatabase.cpp AuditObjectReader.cpp AuditDeployment.cpp AuditJson.cpp)
target_compile_features(etlforge_audit PUBLIC cxx_std_17)
target_include_directories(etlforge_audit PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${ETLFORGE_BOOST_ROOT}")
target_compile_definitions(etlforge_audit PUBLIC NOMINMAX BOOST_ALL_NO_LIB)
target_compile_options(etlforge_audit PRIVATE /utf-8 /W4)
target_link_libraries(etlforge_audit PUBLIC bcrypt odbc32 crypt32 ole32)
add_executable(audit_comparison_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native/audit_comparison_tests.cpp")
target_compile_options(audit_comparison_tests PRIVATE /utf-8)
target_link_libraries(audit_comparison_tests PRIVATE etlforge_audit)
add_executable(audit_database_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native/audit_database_tests.cpp")
target_compile_options(audit_database_tests PRIVATE /utf-8)
target_link_libraries(audit_database_tests PRIVATE etlforge_audit)
enable_testing()
add_test(NAME audit_comparison COMMAND audit_comparison_tests)
add_executable(audit_capture_probe "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native/audit_capture_probe.cpp")
target_link_libraries(audit_capture_probe PRIVATE etlforge_audit)
add_executable(audit_layout_rebaseline "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native/audit_layout_rebaseline.cpp")
target_link_libraries(audit_layout_rebaseline PRIVATE etlforge_audit)
